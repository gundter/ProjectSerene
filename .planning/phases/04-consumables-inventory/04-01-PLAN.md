class PROJECTSERENE_API UItemDataAsset : public UPrimaryDataAsset

    UPROPERTY(EditDefaultsOnly, BlueprintReadOnly, Category = "Item|Display")

    TObjectPtr<UTexture2D> Icon;

    UPROPERTY(EditDefaultsOnly, BlueprintReadOnly, Category = "Item")
    FGameplayTag ItemCategory;

    // Stacking - common items (5), rare items (2) per CONTEXT.md
    UPROPERTY(EditDefaultsOnly, BlueprintReadOnly, Category = "Item")
    int32 MaxStackSize = 5;

    // Effect to apply when used (instant Gameplay Effect)
    UPROPERTY(EditDefaultsOnly, BlueprintReadOnly, Category = "Item|Consumable")
    TSubclassOf<UGameplayEffect> UseEffect;

    // World pickup mesh
    UPROPERTY(EditDefaultsOnly, BlueprintReadOnly, Category = "Item|Pickup")
    TObjectPtr<UStaticMesh> PickupMesh;

    // Asset Manager integration
    virtual FPrimaryAssetId GetPrimaryAssetId() const override;
};

Create UInventoryComponent as ActorComponent for inventory management:

**InventoryComponent.h:**
```cpp
DECLARE_DYNAMIC_MULTICAST_DELEGATE_TwoParams(FOnInventoryChanged, UItemDataAsset*, Item, int32, NewQuantity);

UCLASS(ClassGroup=(Horror), meta=(BlueprintSpawnableComponent))
class PROJECTSERENE_API UInventoryComponent : public UActorComponent
{
    GENERATED_BODY()
public:
    UInventoryComponent();

    // Core operations
    UFUNCTION(BlueprintCallable, Category = "Inventory")
    bool AddItem(UItemDataAsset* Item, int32 Quantity = 1);

    UFUNCTION(BlueprintCallable, Category = "Inventory")
    bool RemoveItem(UItemDataAsset* Item, int32 Quantity = 1);

    UFUNCTION(BlueprintCallable, Category = "Inventory")
    bool UseItem(UItemDataAsset* Item);

    UFUNCTION(BlueprintCallable, Category = "Inventory")
    bool CanAddItem(UItemDataAsset* Item, int32 Quantity = 1) const;

    // Query operations
    UFUNCTION(BlueprintCallable, Category = "Inventory")
    int32 GetItemQuantity(UItemDataAsset* Item) const;

    UFUNCTION(BlueprintCallable, Category = "Inventory")
    TArray<UItemDataAsset*> GetAllItems() const;

    UFUNCTION(BlueprintCallable, Category = "Inventory")
    TArray<UItemDataAsset*> GetItemsByCategory(FGameplayTag Category) const;

    // Quick slots (4 per CONTEXT.md)
    UFUNCTION(BlueprintCallable, Category = "Inventory|QuickSlots")
    void AssignQuickSlot(int32 SlotIndex, UItemDataAsset* Item);

    UFUNCTION(BlueprintCallable, Category = "Inventory|QuickSlots")
    UItemDataAsset* GetQuickSlotItem(int32 SlotIndex) const;

    UFUNCTION(BlueprintCallable, Category = "Inventory|QuickSlots")
    bool UseQuickSlot(int32 SlotIndex);

    // Delegate for UI binding
    UPROPERTY(BlueprintAssignable, Category = "Inventory")
    FOnInventoryChanged OnInventoryChanged;

protected:
    // Item storage: maps item asset to quantity
    UPROPERTY()
    TMap<TObjectPtr<UItemDataAsset>, int32> Items;

    // Quick slots array (fixed size 4)
    UPROPERTY()
    TArray<TObjectPtr<UItemDataAsset>> QuickSlots;

    static const int32 NUM_QUICK_SLOTS = 4;
};
```

**InventoryComponent.cpp:**
Implement all methods. Key implementation notes:

- **AddItem:** Check MaxStackSize, return false if would exceed (per CONTEXT.md "Inventory full" message). Broadcast OnInventoryChanged.
- **UseItem:** Get ASC from owning PlayerState (Cast<ASerenePlayerState>(GetOwner())), apply UseEffect as instant Gameplay Effect using pattern from SerenePlayerState::InitializeAttributes. Decrement quantity. Broadcast OnInventoryChanged.
- **GetItemsByCategory:** Filter Items TMap by checking Item->ItemCategory.MatchesTag(Category).
- **QuickSlots:** Initialize array to NUM_QUICK_SLOTS nullptrs in constructor. ValidateSlotIndex before access.

Use UPROPERTY() on TMap to prevent GC issues (per RESEARCH.md Pitfall 2).
