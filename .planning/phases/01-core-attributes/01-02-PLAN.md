   ```cpp
   #pragma once

   #include "CoreMinimal.h"
   #include "GameFramework/PlayerState.h"
   #include "AbilitySystemInterface.h"
   #include "SerenePlayerState.generated.h"

   class USereneAbilitySystemComponent;
   class USereneAttributeSet;
   class UGameplayEffect;

   UCLASS()
   class PROJECTSERENE_API ASerenePlayerState : public APlayerState, public IAbilitySystemInterface
   {
       GENERATED_BODY()

   public:
       ASerenePlayerState();

       // IAbilitySystemInterface
       virtual UAbilitySystemComponent* GetAbilitySystemComponent() const override;

       // Attribute access
       USereneAttributeSet* GetAttributeSet() const { return AttributeSet; }

       // Initialize attributes with default Gameplay Effect
       void InitializeAttributes();

   protected:
       UPROPERTY()
       TObjectPtr<USereneAbilitySystemComponent> AbilitySystemComponent;

       UPROPERTY()
       TObjectPtr<USereneAttributeSet> AttributeSet;

       // Default Gameplay Effect to initialize attribute values
       UPROPERTY(EditDefaultsOnly, Category = "GAS")
       TSubclassOf<UGameplayEffect> DefaultAttributeEffect;
   };
   ```

4. Create `SerenePlayerState.cpp`:
   - Constructor: CreateDefaultSubobject for ASC and AttributeSet
   - GetAbilitySystemComponent: return ASC
   - InitializeAttributes: Apply DefaultAttributeEffect if valid
     ```cpp
     void ASerenePlayerState::InitializeAttributes()
     {
         if (AbilitySystemComponent && DefaultAttributeEffect)
         {
             FGameplayEffectContextHandle ContextHandle = AbilitySystemComponent->MakeEffectContext();
             FGameplayEffectSpecHandle SpecHandle = AbilitySystemComponent->MakeOutgoingSpec(DefaultAttributeEffect, 1, ContextHandle);
             if (SpecHandle.IsValid())
             {
                 AbilitySystemComponent->ApplyGameplayEffectSpecToSelf(*SpecHandle.Data.Get());
             }
         }
     }
     ```

     ```cpp
     void AHorrorCharacter::PossessedBy(AController* NewController)
     {
         Super::PossessedBy(NewController);

         if (ASerenePlayerState* PS = GetPlayerState<ASerenePlayerState>())
         {
             // Initialize ASC with PlayerState as Owner, Character as Avatar
             PS->GetAbilitySystemComponent()->InitAbilityActorInfo(PS, this);

             // Initialize attributes
             PS->InitializeAttributes();

             // Bind attribute change delegate for stamina
             PS->GetAbilitySystemComponent()->GetGameplayAttributeValueChangeDelegate(
                 USereneAttributeSet::GetStaminaAttribute())
                 .AddUObject(this, &AHorrorCharacter::OnStaminaChanged);

             // Start stamina regen (will be blocked by sprinting tag)
             StartStaminaRegen();
         }
     }
     ```

   - Add OnStaminaChanged handler:
     ```cpp
     void AHorrorCharacter::OnStaminaChanged(const FOnAttributeChangeData& Data)
     {
         if (ASerenePlayerState* PS = GetPlayerState<ASerenePlayerState>())
         {
             float MaxStamina = PS->GetAttributeSet()->GetMaxStamina();
             float Percent = MaxStamina > 0 ? Data.NewValue / MaxStamina : 0;
             OnSprintMeterUpdated.Broadcast(Percent);

             // Handle recovery threshold: can't sprint again until 20% stamina
             if (bRecovering && Data.NewValue >= MaxStamina * 0.2f)
             {
                 bRecovering = false;
             }

             // Force stop sprint if stamina depleted
             if (Data.NewValue <= 0 && bSprinting)
             {
                 DoEndSprint();
                 bRecovering = true;
             }
         }
     }
     ```
